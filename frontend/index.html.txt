        const API_URL = "http://localhost:5000/api/timetable";
function updateSectionVisibility() {

    const branch = document.getElementById('branch').value;
    const section = document.getElementById('section');

    if (branch === 'Civil') {
        section.style.display = 'block';
    } else {
        section.style.display = 'none';
    }
}


        // Load timetable from backend
        async function loadTimetable() {

            const tableBody = document.getElementById("timetableBody");

            tableBody.innerHTML = `
                <tr>
                    <td colspan="6" class="loading">
                        Loading timetable...
                    </td>
                </tr>
            `;

            try {

                const response = await fetch(`${API_URL}?branch=${encodeURIComponent(document.getElementById("branch").value)}&section=${encodeURIComponent(document.getElementById("section").value)}&semester=${encodeURIComponent(document.getElementById("semester").value)}`);

                if (!response.ok) {
                    throw new Error("Server response error");
                }

                const data = await response.json();

                displayTimetable(data);

            } catch (error) {

                console.error(error);

                tableBody.innerHTML = `
                    <tr>
                        <td colspan="6" class="loading">
                            ⚠️ Unable to connect to backend.
                            <br><br>
                            Make sure the backend server is running on port 5000.
                        </td>
                    </tr>
                `;

            }

        }


        // Display timetable
        function displayTimetable(data) {

            const tableBody = document.getElementById("timetableBody");

            document.getElementById("totalClasses").textContent = data.length;

            const faculties = new Set();
            const rooms = new Set();
            const slots = new Set();

            if (data.length === 0) {

                tableBody.innerHTML = `
                    <tr>
                        <td colspan="6" class="loading">
                            No timetable data available.
                            <br><br>
                            Click <b>⚡ Auto Generate</b> after the backend
                            generation API is ready.
                        </td>
                    </tr>
                `;

            } else {

                tableBody.innerHTML = "";

                data.forEach(item => {

                    if (
    item.facultyId?.name &&
    item.branch === document.getElementById("branch").value
) {
    faculties.add(item.facultyId.name);
}

                    if (item.roomId?.roomNo) {
                        rooms.add(item.roomId.roomNo);
                    }

                    if (item.timeSlotId) {
                        slots.add(item.timeSlotId._id || item.timeSlotId);
                    }

                    const row = document.createElement("tr");

                    const slotTimes = {
    1: "10:00 AM – 11:00 AM",
    2: "11:00 AM – 12:00 PM",
    3: "12:00 PM – 1:00 PM",
    4: "2:00 PM – 3:00 PM",
    5: "3:00 PM – 4:00 PM",
    6: "4:00 PM – 5:00 PM"
};

const slotNumber = item.timeSlotId?.slotNumber;

row.innerHTML = `
    <td>
        <strong>${item.timeSlotId?.day || "N/A"}</strong>
        <br>
        Slot ${slotNumber || "N/A"}
        <br>
        <small>${slotTimes[slotNumber] || ""}</small>
    </td>

                        <td class="subject">
                            ${item.subjectName || "N/A"}
                        </td>

                        <td>
                            ${item.branch || "N/A"}
                        </td>

                        <td>
                            ${item.semester || "N/A"}
                        </td>

                        <td>
                            ${item.facultyId?.name || "Unassigned"}
                        </td>

                        <td>
                            <span class="room">
                                ${item.roomId?.roomNo || "TBD"}
                            </span>
                        </td>
                    `;

                    tableBody.appendChild(row);

                });

            }

            document.getElementById("facultyCount").textContent =
                faculties.size;

            document.getElementById("roomCount").textContent =
                rooms.size;

            document.getElementById("slotCount").textContent =
                slots.size;

        }


        // Auto generate timetable
        async function generateTimetable() {

            const button = event.target;

            button.disabled = true;
            button.textContent = "Generating...";

            try {

                const response = await fetch(API_URL + "/generate", {

                    method: "POST",

                    headers: {
                        "Content-Type": "application/json"
                    },

                    body: JSON.stringify({
    branch: document.getElementById("branch").value,
    section: document.getElementById("section").value,
    semester: Number(
        document.getElementById("semester").value
    )
})

                });

                const result = await response.json();

                if (!response.ok) {
                    throw new Error(result.error || result.message);
                }

                alert("Timetable generated successfully!");

                loadTimetable();

            } catch (error) {

                alert(
                    "Generation API is not ready yet.\n\n" +
                    error.message
                );

            }

            button.disabled = false;
            button.textContent = "⚡ Auto Generate";

        }

        loadTimetable();
// Check faculty leave and find substitute
async function checkLeaveRequest() {

    const facultyId =
        document.getElementById("leaveFaculty").value;

    const timeSlotId =
        document.getElementById("leaveTimeSlot").value;

    const leaveDate =
        document.getElementById("leaveDate").value;

    const resultBox =
        document.getElementById("substituteResults");

    if (!facultyId || !timeSlotId || !leaveDate) {
        resultBox.innerHTML =
            "<p style='color:red;'>Please select faculty, time slot and leave date.</p>";
        return;
    }

    resultBox.innerHTML =
        "<p>🔄 Checking available substitutes...</p>";

    try {

        const response = await fetch(
            API_URL + "/leave-request",
            {
                method: "POST",

                headers: {
                    "Content-Type": "application/json"
                },

                body: JSON.stringify({
                    facultyId: facultyId,
                    timeSlotId: timeSlotId,
                    leaveDate: leaveDate
                })
            }
        );

        const result = await response.json();

        if (!response.ok) {
            throw new Error(
                result.message || result.error
            );
        }

        let html = `
            <h3>Available Substitute Faculty</h3>
            <p>
                <strong>Subject:</strong>
                ${result.subjectName}
            </p>
        `;

        result.substituteOptions.forEach(function (faculty) {

            html += `
                <div style="
                    padding:10px;
                    margin:8px 0;
                    border:1px solid #ccc;
                    border-radius:8px;
                ">
                    👨‍🏫 <strong>${faculty.name}</strong>
                    <br>
                    <small>${faculty.type}</small>
                </div>
            `;

        });

        resultBox.innerHTML = html;

    } catch (error) {

        resultBox.innerHTML = `
            <p style="color:red;">
                ❌ ${error.message}
            </p>
        `;
    }
}
